# Найти пакет Protobuf
find_package(Protobuf REQUIRED)

# Поиск gRPC
find_package(gRPC REQUIRED)

# Найти плагин gRPC
find_program(GRPC_CPP_PLUGIN 
    NAMES grpc_cpp_plugin
    HINTS ${gRPC_ROOT} ${gRPC_DIR}
    PATHS /usr/bin /usr/local/bin /opt/homebrew/bin
    REQUIRED
)
message(STATUS "GRPC plugin found at: ${GRPC_CPP_PLUGIN}")

# Перечисление proto файлов
set(REGULAR_PROTO_FILES
    user.proto
    product.proto
    anydata.proto
    storage.proto
)

# Генерируем файлы .pb.cc и .pb.h 
protobuf_generate_cpp(REGULAR_SRCS REGULAR_HEADERS ${REGULAR_PROTO_FILES}
    PROTOC_OUT_DIR ${CMAKE_BINARY_DIR}/protos 
)

# Сервисный proto файл (с gRPC)
set(SERVICE_PROTO_FILES nosql_service.proto)

# Генерация protobuf + gRPC для сервиса
protobuf_generate_cpp(SERVICE_SRCS SERVICE_HEADERS ${SERVICE_PROTO_FILES}
    PROTOC_OUT_DIR ${CMAKE_BINARY_DIR}/protos
    GRPC
    PLUGIN "protoc-gen-grpc=${GRPC_CPP_PLUGIN}" 
)

# Fallback если генерация не сработала
if(NOT SERVICE_SRCS OR NOT EXISTS "${CMAKE_BINARY_DIR}/protos/nosql_service.grpc.pb.cc")
    message(WARNING "GRPC files not generated by protobuf_generate_cpp, using manual generation")
    
    set(SERVICE_PB_SRCS "${CMAKE_BINARY_DIR}/protos/nosql_service.pb.cc")
    set(SERVICE_PB_HDRS "${CMAKE_BINARY_DIR}/protos/nosql_service.pb.h")
    set(GRPC_SRCS "${CMAKE_BINARY_DIR}/protos/nosql_service.grpc.pb.cc")
    set(GRPC_HDRS "${CMAKE_BINARY_DIR}/protos/nosql_service.grpc.pb.h")
    
    add_custom_command(
        OUTPUT ${GRPC_SRCS} ${GRPC_HDRS} ${SERVICE_PB_SRCS} ${SERVICE_PB_HDRS}
        COMMAND ${Protobuf_PROTOC_EXECUTABLE}
        ARGS
            --cpp_out=${CMAKE_BINARY_DIR}/protos
            --grpc_out=${CMAKE_BINARY_DIR}/protos
            --plugin=protoc-gen-grpc=${GRPC_CPP_PLUGIN}
            -I ${CMAKE_CURRENT_SOURCE_DIR}
            ${CMAKE_CURRENT_SOURCE_DIR}/nosql_service.proto
        DEPENDS 
            ${CMAKE_CURRENT_SOURCE_DIR}/nosql_service.proto
            ${GRPC_CPP_PLUGIN}
        COMMENT "Manually generating gRPC code for nosql_service.proto"
        VERBATIM
    )
    
    set(SERVICE_SRCS ${GRPC_SRCS} ${SERVICE_PB_SRCS})
endif()

# Собрать все файлы
set(ALL_SRCS ${REGULAR_SRCS} ${SERVICE_SRCS})

# Создаем библиотеку
add_library(protos_tool STATIC ${ALL_SRCS})

set_target_properties(protos_tool PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Добавляем инклюды
target_include_directories(protos_tool PUBLIC
    ${CMAKE_BINARY_DIR}/protos
)

# Линкуем сами генерируемые файлы с gRPC
target_link_libraries(protos_tool PUBLIC
    gRPC::grpc++
    gRPC::grpc
    protobuf::libprotobuf
)
